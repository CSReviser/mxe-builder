bool Utility::isSameOrSubDirectory(
    const QString &target,
    const QString &parent)
{
    if (target.isEmpty() || parent.isEmpty())
        return false;

#ifdef Q_OS_WIN
    const Qt::CaseSensitivity cs = Qt::CaseInsensitive;
#else
    const Qt::CaseSensitivity cs = Qt::CaseSensitive;
#endif

    if (target.compare(parent, cs) == 0)
        return true;

    QString prefix = parent;

    if (!prefix.endsWith('/'))
        prefix += '/';

    return target.startsWith(prefix, cs);
}

QString resolveIniPath()
{
    const QString appDir  = QCoreApplication::applicationDirPath();
    const QString iniName = "CaptureStream2.ini";
    const QString portableIni = QDir(appDir).filePath(iniName);

    // exe横に実際に書き込めるなら従来どおり（既存ini有無を問わない）

    if (QFileInfo::exists(portableIni) && !QFileInfo::exists(fallbackIni))
        QFile::copy(portableIni, fallbackIni);

    return fallbackIni;
}

QString resolveIniPath()
{
    const QString appDir  = QCoreApplication::applicationDirPath();
    const QString iniName = "CaptureStream2.ini";
    const QString portableIni = QDir(appDir).filePath(iniName);

    // 1. すでにexe横にiniが存在する場合
    if (QFileInfo::exists(portableIni)) {
        // 書き込み可能ならそのまま使う
        if (QFileInfo(portableIni).isWritable() && !isInProtectedDir(appDir)) {
            return portableIni;
        }
        // 書き込み不可（Program Files等）であっても、既存設定を読むために採用する
        // ※ ただし、将来の書き込みのためにAppDataへ移行する処理を挟むと完璧です
        return portableIni;
    }

    // 2. 新規作成の場合：exe横に書き込めるかチェック
    // （保護フォルダ内でない かつ 実際に書き込みテストが成功する場合のみポータブル）
    if (!isInProtectedDir(appDir) && isDirWritable(appDir)) {
        return portableIni;
    }

    // 3. フォールバック (%APPDATA%など)
    const QString cfgDir = QStandardPaths::writableLocation(QStandardPaths::AppConfigLocation);
    QDir().mkpath(cfgDir);
    const QString fallbackIni = QDir(cfgDir).filePath(iniName);

    // ★もし「exe横には書き込めないが、古いポータブルiniがたまたま存在していた場合」の救済措置
    // （既存の portableIni があって、かつ fallbackIni がまだ無いならコピーしてあげる）
    if (QFileInfo::exists(portableIni) && !QFileInfo::exists(fallbackIni)) {
        QFile::copy(portableIni, fallbackIni);
    }

    return fallbackIni;
}

void Settings::initializeDefaults()
{
    QSettings settings = createSettings();
    settings.beginGroup(Customize::SETTING_GROUP);

    for (int i = 0; i < Customize::COURSES.size(); ++i) {
        if (!settings.contains(Customize::TITLE_KEYS[i]))

        if (!settings.contains(Customize::FILENAME_KEYS[i]))

}

void CustomizeDialog::loadSettings()
{
    ui.lineEdit->setText(
        mode == Ui::TitleMode
            ? Settings::titleFormat(0)
            : Settings::fileNameFormat(0)
    );

    ui.lineEdit_2->setText(
        mode == Ui::TitleMode
            ? Settings::titleFormat(1)
            : Settings::fileNameFormat(1)
    );

    if (mode == Ui::TitleMode)
        ui.checkBox->setChecked(Settings::tagSpaceFlag());
    else
        ui.checkBox->setChecked(Settings::nameSpaceFlag());
}


void CustomizeDialog::saveSettings()
{
    if (mode == Ui::TitleMode) {
        Settings::setTitleFormat(0, ui.lineEdit->text());
        Settings::setTitleFormat(1, ui.lineEdit_2->text());
        Settings::setTagSpaceFlag(ui.checkBox->isChecked());
    } else {
        Settings::setFileNameFormat(0, ui.lineEdit->text());
        Settings::setFileNameFormat(1, ui.lineEdit_2->text());
        Settings::setNameSpaceFlag(ui.checkBox->isChecked());
    }
}
